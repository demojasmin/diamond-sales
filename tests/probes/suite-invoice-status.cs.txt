using System.Reflection;
using System.Threading;
using System.Windows;
using System.Windows.Controls;
using DiamondDesktop;
using DiamondDesktop.Data;

int fail = 0;
void Check(string n, bool ok, string? d = null)
{ Console.WriteLine($"{(ok ? "PASS" : "FAIL")}  {n}{(d is null ? "" : $"  . {d}")}"); if (!ok) fail++; }

var t = new Thread(() =>
{
    SynchronizationContext.SetSynchronizationContext(
        new System.Windows.Threading.DispatcherSynchronizationContext(
            System.Windows.Threading.Dispatcher.CurrentDispatcher));
    var app = new Application();
    foreach (string d in new[]{"Themes/LightTheme.xaml","Themes/Typography.xaml","Styles/Buttons.xaml",
        "Styles/Inputs.xaml","Styles/Widgets.xaml","Styles/DataGridStyles.xaml","Styles/AppOverrides.xaml",
        "Styles/Components.xaml","Styles/Dashboard.xaml","Styles/Audit.xaml", "Styles/Catalogue.xaml"})
        app.Resources.MergedDictionaries.Add(new ResourceDictionary
        { Source = new Uri($"pack://application:,,,/DiamondDesktop;component/{d}") });

    var win = new MainWindow { WindowStartupLocation = WindowStartupLocation.Manual,
        Left = -4000, Top = -4000, ShowInTaskbar = false, Width = 1420 };
    win.Show();

    // The STATUS filter is gone from this page on request, and the DATE control that replaced it
    // is the SAME AppDatePicker Sales entry uses -- one date field in the app, one way to pick a
    // date. What went with the status filter: Sale / Cancelled and the three payment states. The
    // STATUS column still SHOWS each row's state; nothing narrows by it any more.
    var filter = (DatePicker)win.FindName("InvoiceDateFilter")!;

    Check("LIST - the status filter is gone and a date picker stands in its place",
          win.FindName("InvoiceStatusFilter") is null && filter is not null);
    // LineFilterDate, not InvoiceDatePicker. The entry's own date box is off the screen -- it was
    // the SECOND date field on that page, four inches from the filter -- so the date control Sales
    // entry shows is its filter, and these two are the pair that must match.
    Check("LIST - it is the same date control Sales entry uses",
          filter.Style == win.FindResource("AppDatePicker")
          && ((DatePicker)win.FindName("LineFilterDate")!).Style == win.FindResource("AppDatePicker")
          && ((DatePicker)win.FindName("LineFilterDate")!).Width == filter.Width);
    Check("LIST - and it opens empty, which means every date",
          filter.SelectedDate is null, filter.SelectedDate?.ToString() ?? "<empty>");

    // Dated relative to today, so the fixture moves with the clock rather than rotting.
    var today = DateOnly.FromDateTime(DateTime.Today);
    typeof(MainWindow).GetField("_invoices", BindingFlags.Instance | BindingFlags.NonPublic)!
        .SetValue(win, new List<VInvoice>
        {
            new() { InvoiceId = 1, InvoiceNo = "INV-1", BuyerName = "ABC", Status = InvoiceStatus.POSTED,
                    InvoiceDate = today, AmountTotal = 100m, Outstanding = 100m },
            new() { InvoiceId = 2, InvoiceNo = "INV-2", BuyerName = "ABC", Status = InvoiceStatus.POSTED,
                    InvoiceDate = today.AddDays(-3), AmountTotal = 100m, Outstanding = 0m },
            new() { InvoiceId = 3, InvoiceNo = "INV-3", BuyerName = "ABC", Status = InvoiceStatus.CANCELLED,
                    InvoiceDate = today.AddDays(-200) },
            // A memo dated today, which must never appear whatever the date says.
            new() { InvoiceId = 4, InvoiceNo = null, BuyerName = "ABC", Status = InvoiceStatus.DRAFT,
                    InvoiceDate = today },
        });
    var apply = typeof(MainWindow).GetMethod("ApplyInvoiceFilter",
        BindingFlags.Instance | BindingFlags.NonPublic)!;
    var grid = (DataGrid)win.FindName("InvoiceGrid")!;
    List<VInvoice> Shown(DateOnly? on)
    {
        filter.SelectedDate = on?.ToDateTime(TimeOnly.MinValue);
        apply.Invoke(win, null);
        return ((IEnumerable<VInvoice>)grid.ItemsSource).ToList();
    }

    Check("FILTER - no date chosen shows every sale, and no memo",
          Shown(null).Count == 3 && Shown(null).All(i => i.Status != InvoiceStatus.DRAFT),
          string.Join(", ", Shown(null).Select(i => i.InvoiceNo)));
    Check("FILTER - a date narrows to that day, and the memo dated the same day stays out",
          Shown(today).Count == 1 && Shown(today)[0].InvoiceNo == "INV-1",
          string.Join(", ", Shown(today).Select(i => i.InvoiceNo)));
    Check("FILTER - another day gives that day's sale",
          Shown(today.AddDays(-3)).Count == 1 && Shown(today.AddDays(-3))[0].InvoiceNo == "INV-2");
    Check("FILTER - a day with nothing on it comes back empty rather than showing everything",
          Shown(today.AddDays(-1)).Count == 0, $"{Shown(today.AddDays(-1)).Count}");

    // ONLY CONFIRMED SALES REACH THIS PAGE. Drafts are filtered out before the date is applied at
    // all, so no setting can bring an unconfirmed entry back.
    foreach (var day in new DateOnly?[] { null, today, today.AddDays(-3), today.AddDays(-200) })
        if (Shown(day).Any(i => i.Status == InvoiceStatus.DRAFT))
        { Check("FILTER - no date setting brings an unconfirmed entry back", false); break; }
    Check("FILTER - no date setting brings an unconfirmed entry back", true);

    Console.WriteLine();
    Console.WriteLine(fail == 0 ? "All checks passed." : $"{fail} FAILED");
    Environment.Exit(fail);
});
t.SetApartmentState(ApartmentState.STA); t.Start(); t.Join();
