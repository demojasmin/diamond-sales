// The printed approval note, as it actually lays out.
//
// NO DATABASE. Reports.BuildApprovalNote takes two views and a letterhead and returns a
// FlowDocument, so the one artefact a customer physically receives can be checked without a
// printer attached -- which is the whole reason BuildInvoice was split out in the first place.
//
// The structural checks matter more than the textual ones here. A missing word is obvious the
// first time somebody prints; a table that has silently collapsed its columns to two pixels, or a
// note that has spilled onto a second sheet, is not.
using System.Threading;
using System.Windows;
using System.Windows.Documents;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using System.Windows.Controls;
using DiamondDesktop;
using DiamondDesktop.Data;

int fail = 0;
void Check(string n, bool ok, string? d = null)
{
    Console.WriteLine($"{(ok ? "PASS" : "FAIL")}  {n}{(d is null ? "" : $"  . {d}")}");
    if (!ok) fail++;
}

var t = new Thread(() =>
{
    var invoice = new VInvoice
    {
        InvoiceId = 4, InvoiceNo = "INV-2026-00004", InvoiceDate = new DateOnly(2026, 8, 21),
        BuyerId = 11, BuyerName = "QUEST DIAMOND", BrokerName = "R K BROKER",
        BrokerPct = 1m, TermsDays = 45, DocType = "BILL", Status = "POSTED",
        AmountTotal = 139864.73m, CaratsSold = 2.30m,
    };
    var lines = new List<VSalesLine>
    {
        new() { GradeCode = "NO II", SizeCode = "-6.5", GrossWeightCt = 2.30m, SelectionCt = 2.30m,
                PricePerCt = 63000m, ExRate = 1m, Amount = 139864.73m, Remark = "sorted parcel" },
        new() { GradeCode = "NO 1 BB", SizeCode = "+6.5", GrossWeightCt = 12.50m, SelectionCt = 10.25m,
                PricePerCt = 36000m, ExRate = 1m, Amount = 354423.76m, Remark = "" },
    };

    var head = new Reports.Letterhead();
    var doc = Reports.BuildApprovalNote(invoice, lines, head);
    string text = new TextRange(doc.ContentStart, doc.ContentEnd).Text;

    // -- the letterhead --------------------------------------------------------
    Check("NOTE - says what the document is", text.Contains("APPROVAL NOTE"));
    Check("NOTE - carries the company name", text.Contains("PRIYA GEMS"));
    Check("NOTE - and the trade line", text.Contains("MANUFACTURERS OF DIAMONDS"));
    Check("NOTE - the address and the email", text.Contains("Dahisar (E), Mumbai - 400 068.")
                                              && text.Contains("priyagems13@gmail.com"));
    Check("NOTE - the telephone and O.BC", text.Contains("3392 7439") && text.Contains("O.BC : 7439"));

    // -- who it is for ---------------------------------------------------------
    Check("NOTE - names the buyer on the To line", text.Contains("QUEST DIAMOND"));
    Check("NOTE - names the broker on the Through line", text.Contains("R K BROKER"));
    // The pad has NO invoice-number field, so neither does this. The number this app assigns is
    // its own; printing it would put a figure on the note the office has never seen on one.
    Check("NOTE - carries no invoice number, as the pad does not",
          !text.Contains("INV-2026-00004") && !text.Contains("No."),
          text.Contains("INV-2026-00004") ? "the number is on it" : "clean");

    // ── whose note it is, and whether it can be read ───────────────────────
    // Both callers used to hand PrintInvoice the APPLICATION's title, so the note went out headed
    // "Diamond Sales & Inventory" over Priya Gems' address, telephone, email and GSTIN. A
    // letterhead is one identity; the name cannot come from somewhere else than the address.
    Check("REF - headed with the letterhead's own name",
          text.Contains("PRIYA GEMS") && !text.Contains("Diamond Sales & Inventory"),
          text.Contains("Diamond Sales & Inventory") ? "the app title is on it" : "PRIYA GEMS");

    // The note is read across a counter. Nothing on it may be smaller than this.
    var tooSmall = new List<string>();
    void Walk(DependencyObject? d)
    {
        if (d is null) return;
        if (d is System.Windows.Documents.TextElement te && te.FontSize < 11
            && te is not System.Windows.Documents.Span)
            tooSmall.Add($"{te.GetType().Name} @ {te.FontSize}");
        foreach (object child in System.Windows.LogicalTreeHelper.GetChildren(d))
            if (child is DependencyObject dep) Walk(dep);
    }
    foreach (var b in doc.Blocks) Walk(b);
    Check("REF - nothing on the note is printed below 11pt",
          tooSmall.Count == 0, tooSmall.Count == 0 ? "all legible" : string.Join(", ", tooSmall.Take(4)));

    // One sheet. The pad is one sheet, and a note continuing overleaf is a note whose second half
    // goes missing between the counter and the safe.
    var sheet = ((System.Windows.Documents.IDocumentPaginatorSource)doc).DocumentPaginator;
    sheet.ComputePageCount();
    Check("REF - and it fits on ONE page", sheet.PageCount == 1, $"{sheet.PageCount} page(s)");

    // ── everything the reference prints, in the order it prints it ──────────
    string[] mustSay =
    [
        "APPROVAL NOTE",
        "Tel.: 3392 7439",
        "O.BC : 7439",
        "PRIYA GEMS",
        "IMPORTERS",
        "EXPORTERS",
        "MANUFACTURERS OF DIAMONDS",
        "A-101/B, Tanvi's Diamond Ind. Premises Estate, S.V. Road, Dahisar (E), Mumbai - 400 068.",
        "Email : priyagems13@gmail.com",
        "Date :",
        "To,",
        "Through",
        "Mob.:",
        "Please receive the following goods on approval for Export / Local Sale / Assortment / Mfgr.",
        "S.",
        "No",
        "PARTICULARS",
        "WEIGHT",
        "RATE per",
        "Carat",
        "\u20b9 / $",
        "REMARKS",
        "Acknowledgement of entrustment as per the",
        "conditions of reserve.",
        "Subject to Mumbai Jurisdiction.",
        "GSTIN : 27AGRPB2357E1ZH",
        "For PRIYA GEMS",
        "Thru / Receiver's Sign",
        "Authorised Sign.",
    ];
    foreach (string phrase in mustSay)
        Check($"REF - prints \"{phrase}\"", text.Contains(phrase));

    // In the order the pad has them, not merely present somewhere.
    // Searched FORWARD from the last match, not from the top. "S." occurs in "S.V. Road" long
    // before the table header, so an absolute IndexOf reported the header as out of order when it
    // was simply preceded by an address.
    int at = 0;
    bool ordered = true;
    string? outOfOrder = null;
    foreach (string phrase in mustSay)
    {
        int i = text.IndexOf(phrase, at, StringComparison.Ordinal);
        if (i < 0) { ordered = false; outOfOrder = phrase; break; }
        at = i + 1;
    }
    Check("REF - and in the reference's own order", ordered, outOfOrder ?? "in order");


    // A slash inside a date format string is the CULTURE's separator, not a slash. On a machine set
    // to dd-MM-yyyy this printed 21 - 08 - 2026 until the format was escaped.
    Check("NOTE - the date reads as the pad prints it, with slashes",
          text.Contains("21  /  08  /  2026"),
          text.Split('\n').FirstOrDefault(l => l.Contains("Date")) ?? "<no date line>");

    Check("NOTE - the approval wording is intact",
          text.Contains("goods on approval for Export / Local Sale / Assortment / Mfgr."));

    // -- the goods -------------------------------------------------------------
    Check("NOTE - every column heading", text.Contains("PARTICULARS") && text.Contains("WEIGHT")
                                         && text.Contains("RATE per") && text.Contains("REMARKS"));
    Check("NOTE - prints each line's grade and size",
          text.Contains("NO II") && text.Contains("-6.5")
          && text.Contains("NO 1 BB") && text.Contains("+6.5"));
    Check("NOTE - prints the weights", text.Contains("2.30") && text.Contains("12.50"));
    Check("NOTE - prints the rate per carat", text.Contains("63,000.00") && text.Contains("36,000.00"));
    Check("NOTE - prints the per-line remark, which no screen shows", text.Contains("sorted parcel"));

    // -- the acknowledgement ---------------------------------------------------
    Check("NOTE - the entrustment wording", text.Contains("Acknowledgement of entrustment"));
    Check("NOTE - the jurisdiction", text.Contains("Subject to Mumbai Jurisdiction."));
    Check("NOTE - the GSTIN", text.Contains("GSTIN : 27AGRPB2357E1ZH"));
    Check("NOTE - both signature captions",
          text.Contains("Receiver") && text.Contains("Authorised Sign."));
    Check("NOTE - and who it is signed for", text.Contains("For PRIYA GEMS"));

    // -- structure, which is what silently goes wrong --------------------------
    var paginator = (DynamicDocumentPaginator)((IDocumentPaginatorSource)doc).DocumentPaginator;
    paginator.PageSize = new Size(794, 1123);
    paginator.ComputePageCount();

    Check("NOTE - fits on ONE A4 sheet", paginator.PageCount == 1, $"{paginator.PageCount} page(s)");
    Check("NOTE - is A4, not whatever the printer reports",
          doc.PageWidth == 794 && doc.PageHeight == 1123, $"{doc.PageWidth} x {doc.PageHeight}");

    // The defect that printed one character per line: an unbounded content width leaves every star
    // column resolving to nothing. Both of these have to hold for the table to lay out at all.
    Check("NOTE - the content width is bounded, never infinity",
          !double.IsInfinity(doc.ColumnWidth), doc.ColumnWidth.ToString());
    Check("NOTE - the text column may not grow past it", !doc.IsColumnWidthFlexible);

    var table = doc.Blocks.OfType<Table>().Single(x => x.Columns.Count == 5);

    // -- every TYPED value prints in the data colour, every label does not -----
    // The broker slipped through the first time this was done: it was interpolated into its own
    // label ("Through  R K BROKER") before it reached the colouring, so it arrived as one string
    // and printed as stationery while every other value printed as data. Nothing caught it but a
    // person holding the page. This walks the finished document and checks each one by name.
    var runs = new List<Run>();
    void Collect(DependencyObject node)
    {
        foreach (object child in LogicalTreeHelper.GetChildren(node))
        {
            if (child is Run r && r.Text.Trim().Length > 0) runs.Add(r);
            if (child is DependencyObject d) Collect(d);
        }
    }
    Collect(doc);

    Color Of(Run r) => (r.Foreground as SolidColorBrush)?.Color ?? Colors.Transparent;
    Run? Find(string needle) => runs.FirstOrDefault(r => r.Text.Contains(needle));

    foreach (var (what, needle) in new[]
    {
        ("the buyer",   "QUEST DIAMOND"),
        ("the broker",  "R K BROKER"),
        ("the date",    "2026"),
        ("a grade",     "NO II"),
        ("a weight",    "2.30"),
        ("a rate",      "63,000.00"),
        ("a remark",    "sorted parcel"),
    })
    {
        var run = Find(needle);
        Check($"INK - {what} prints as a typed value, not as stationery",
              run is not null && Of(run) != Color.FromRgb(0x1B, 0x2A, 0x6B),
              run is null ? "<not on the note>" : Of(run).ToString());
    }

    foreach (var (what, needle) in new[]
    {
        ("To,",        "To,"),
        ("Through",    "Through"),
        ("Mob.:",      "Mob.:"),
        ("PARTICULARS","PARTICULARS"),
        ("a row rule", "11"),
    })
    {
        var run = Find(needle);
        Check($"INK - {what} stays stationery",
              run is not null && Of(run) == Color.FromRgb(0x1B, 0x2A, 0x6B),
              run is null ? "<missing>" : Of(run).ToString());
    }

    // -- the letterhead's mark actually loaded ---------------------------------
    // Reports.LogoOrNull swallows a load failure and returns null on purpose, so the note still
    // prints when the resource is missing. That is the right behaviour and it is also silent: the
    // note goes out with an empty corner and nobody finds out. These are the checks that do.
    var lockup = doc.Blocks.OfType<Table>().Single(x => x.Columns.Count == 3
                                                     && x.Columns[0].Width.Value == 128);
    var mark = lockup.RowGroups[0].Rows[0].Cells[0].Blocks.FirstOrDefault();
    var markImage = (mark as BlockUIContainer)?.Child as Image;
    Check("LOGO - the mark is the real logo, not a placeholder",
          markImage?.Source is not null, mark?.GetType().Name ?? "<nothing>");
    Check("LOGO - sized 128 square, as the reference sets it",
          markImage is { Width: 128, Height: 128 },
          markImage is null ? "<no image>" : $"{markImage.Width} x {markImage.Height}");
    Check("LOGO - and the same mark sits behind the table as a watermark",
          table.Background is ImageBrush, table.Background?.GetType().Name ?? "<none>");

    // -- the watermark brush must FILL its box, or the note does not print -----
    // This is the check that earns its keep. A brush painting only part of its bounding box -- a
    // Viewport, a TileMode, anything less than the whole area -- makes the Microsoft Print to PDF
    // driver write a ZERO-BYTE file. Nothing throws, the status bar still says "Sent to the
    // printer", the on-screen render is perfect and every other check here passes. It is only
    // visible on paper, which is the one place this document exists.
    var brush = table.Background as ImageBrush;
    Check("PRINT - the watermark brush fills its box (a Viewport prints an EMPTY pdf)",
          brush is { Opacity: 1 }
            && brush.Viewport == new Rect(0, 0, 1, 1)
            && brush.ViewportUnits == BrushMappingMode.RelativeToBoundingBox
            && brush.TileMode == TileMode.None,
          brush is null ? "<no brush>"
            : $"opacity {brush.Opacity}, viewport {brush.Viewport} ({brush.ViewportUnits}), tile {brush.TileMode}");
    Check("PRINT - and carries no alpha channel for a driver to misread",
          brush?.ImageSource is BitmapSource { Format.Masks.Count: 3 },
          (brush?.ImageSource as BitmapSource)?.Format.ToString() ?? "<none>");
    Check("NOTE - the goods table has five columns", table.Columns.Count == 5);
    Check("NOTE - every column has a real width, none left to a star",
          table.Columns.All(c => c.Width.IsAbsolute && c.Width.Value > 20),
          string.Join(" | ", table.Columns.Select(c => c.Width.Value.ToString())));
    // 8mm a side, as the reference sets it: 794 - 2 x (8 x 96 / 25.4).
    Check("NOTE - the columns total the printable width",
          Math.Abs(table.Columns.Sum(c => c.Width.Value) - (794 - (2 * 8 * 96 / 25.4))) < 1,
          table.Columns.Sum(c => c.Width.Value).ToString());

    // -- the pad is ruled whether or not the lines are used --------------------
    int rows = table.RowGroups[0].Rows.Count;
    Check("NOTE - eleven ruled rows plus a heading", rows == 12, $"{rows} row(s)");

    var one = Reports.BuildApprovalNote(invoice, [lines[0]], head);
    var oneTable = one.Blocks.OfType<Table>().Single(x => x.Columns.Count == 5);
    Check("NOTE - a one-line note is still ruled to eleven",
          oneTable.RowGroups[0].Rows.Count == 12, $"{oneTable.RowGroups[0].Rows.Count} row(s)");

    // More lines than the pad has rules must not be dropped to make it fit.
    var many = Enumerable.Range(0, 18).Select(i => new VSalesLine
    {
        GradeCode = "NO II", SizeCode = "-6.5", GrossWeightCt = 1m, PricePerCt = 1000m, ExRate = 1m,
        Remark = $"row{i}",
    }).ToList();
    var big = Reports.BuildApprovalNote(invoice, many, head);
    string bigText = new TextRange(big.ContentStart, big.ContentEnd).Text;
    Check("NOTE - eighteen lines all print, none dropped to fit the pad",
          Enumerable.Range(0, 18).All(i => bigText.Contains($"row{i}")));

    // -- the name follows the database, the stationery does not ----------------
    var other = Reports.BuildApprovalNote(invoice, lines, head with { Name = "PRIYA SALES" });
    string otherText = new TextRange(other.ContentStart, other.ContentEnd).Text;
    Check("NOTE - the company name is the one passed in",
          otherText.Contains("PRIYA SALES") && otherText.Contains("For PRIYA SALES"));
    Check("NOTE - and the rest of the letterhead is unchanged by it",
          otherText.Contains("GSTIN : 27AGRPB2357E1ZH") && otherText.Contains("3392 7439"));

    // ── ONE NOTE PER BUYER, IN ONE DOCUMENT ──────────────────────────────────
    // It used to be one print job per buyer: two dialogs, two documents, two things to collect
    // from the printer and no way to keep them as a single PDF. One document now, a page break
    // before every note after the first, in selection order.
    var abcLines = new List<VSalesLine>
    {
        new() { GradeCode = "1BB", SizeCode = "+6.5", GrossWeightCt = 1.00m, PricePerCt = 1000m,
                ExRate = 1m, Remark = "abcOnly" },
    };
    var xyzLines = new List<VSalesLine>
    {
        new() { GradeCode = "NO 2", SizeCode = "-6.5", GrossWeightCt = 2.00m, PricePerCt = 2000m,
                ExRate = 1m, Remark = "xyzOnly" },
        new() { GradeCode = "NO 2", SizeCode = "+6.5", GrossWeightCt = 3.00m, PricePerCt = 3000m,
                ExRate = 1m, Remark = "xyzAlso" },
    };
    var abcNote = new VInvoice { InvoiceNo = "MEMO", InvoiceDate = invoice.InvoiceDate,
                                 BuyerName = "ABC EXPORTS", BrokerName = "JITESH SHAH" };
    var xyzNote = new VInvoice { InvoiceNo = "MEMO", InvoiceDate = invoice.InvoiceDate,
                                 BuyerName = "XYZ GEMS", BrokerName = "RAJ BROKERS" };

    // ONE BUYER IS EXACTLY WHAT IT WAS. No wrapper, no break, one page -- the single-buyer case
    // must not pay anything for the multi-buyer one.
    var solo = Reports.BuildApprovalNotes([(abcNote, abcLines)], head);
    var soloPager = (DynamicDocumentPaginator)((IDocumentPaginatorSource)solo).DocumentPaginator;
    soloPager.PageSize = new Size(794, 1123);
    soloPager.ComputePageCount();
    Check("MULTI - one buyer is still one page", soloPager.PageCount == 1, $"{soloPager.PageCount}");
    Check("MULTI - and still the same document the template builds",
          solo.Blocks.OfType<Section>().All(b => !b.BreakPageBefore)
          && solo.PageWidth == 794 && solo.PageHeight == 1123,
          "no page break, same A4 box");

    var both = Reports.BuildApprovalNotes([(abcNote, abcLines), (xyzNote, xyzLines)], head);
    var bothPager = (DynamicDocumentPaginator)((IDocumentPaginatorSource)both).DocumentPaginator;
    bothPager.PageSize = new Size(794, 1123);
    bothPager.ComputePageCount();
    Check("MULTI - two buyers make two pages, one each",
          bothPager.PageCount == 2, $"{bothPager.PageCount} page(s)");
    Check("MULTI - with a page break before the second, not a run-on",
          both.Blocks.OfType<Section>().Count(b => b.BreakPageBefore) == 1,
          $"{both.Blocks.OfType<Section>().Count(b => b.BreakPageBefore)} break(s)");
    Check("MULTI - and the document is still A4, from the note's own template",
          both.PageWidth == 794 && both.PageHeight == 1123 && !both.IsColumnWidthFlexible
          && !double.IsInfinity(both.ColumnWidth));

    // THE WHOLE TEMPLATE ON EVERY PAGE, not a header on the first and rows after it.
    string bothText = new TextRange(both.ContentStart, both.ContentEnd).Text;
    foreach (string stamp in new[] { "APPROVAL NOTE", "PRIYA GEMS", "GSTIN : 27AGRPB2357E1ZH",
                                    "Authorised Sign.", "PARTICULARS" })
        Check($"MULTI - \"{stamp}\" appears once per buyer",
              System.Text.RegularExpressions.Regex.Matches(
                  bothText, System.Text.RegularExpressions.Regex.Escape(stamp)).Count >= 2,
              $"{System.Text.RegularExpressions.Regex.Matches(bothText, System.Text.RegularExpressions.Regex.Escape(stamp)).Count}");

    Check("MULTI - both buyers are named, each on their own note",
          bothText.Contains("ABC EXPORTS") && bothText.Contains("XYZ GEMS"));
    Check("MULTI - and each broker with them",
          bothText.Contains("JITESH SHAH") && bothText.Contains("RAJ BROKERS"));

    // NOTHING MIXES. The lists were divided before they reached the printer, and the pages have to
    // show it: each buyer's page is ruled to eleven and carries only that buyer's parcels.
    var pages = both.Blocks.OfType<Table>().Where(x => x.Columns.Count == 5).ToList();
    var wrapped = both.Blocks.OfType<Section>()
                      .SelectMany(sec => sec.Blocks.OfType<Table>())
                      .Where(x => x.Columns.Count == 5);
    pages.AddRange(wrapped);
    Check("MULTI - one ruled table per buyer", pages.Count == 2, $"{pages.Count}");

    string Rows(Table x) => new TextRange(x.ContentStart, x.ContentEnd).Text;
    Check("MULTI - the first note carries ABC's line and nothing of XYZ's",
          Rows(pages[0]).Contains("abcOnly")
          && !Rows(pages[0]).Contains("xyzOnly") && !Rows(pages[0]).Contains("xyzAlso"),
          "no mixing");
    Check("MULTI - the second carries XYZ's two lines and nothing of ABC's",
          Rows(pages[1]).Contains("xyzOnly") && Rows(pages[1]).Contains("xyzAlso")
          && !Rows(pages[1]).Contains("abcOnly"),
          "no mixing");
    Check("MULTI - and each is ruled to eleven, as one buyer's note always was",
          pages.All(x => x.RowGroups[0].Rows.Count == 12),
          string.Join(", ", pages.Select(x => x.RowGroups[0].Rows.Count)));

    // SELECTION ORDER, not sorted and not reversed: the paper is handed over in the order the
    // deals appear on the entry.
    var reversed = Reports.BuildApprovalNotes([(xyzNote, xyzLines), (abcNote, abcLines)], head);
    string firstOf(FlowDocument d) =>
        new TextRange(d.ContentStart, d.ContentEnd).Text.Contains("XYZ GEMS")
        && new TextRange(d.ContentStart, d.ContentEnd).Text.IndexOf("XYZ GEMS", StringComparison.Ordinal)
           < new TextRange(d.ContentStart, d.ContentEnd).Text.IndexOf("ABC EXPORTS", StringComparison.Ordinal)
            ? "XYZ" : "ABC";
    Check("MULTI - the pages come out in the order they were given",
          firstOf(both) == "ABC" && firstOf(reversed) == "XYZ",
          $"{firstOf(both)} then, reversed, {firstOf(reversed)}");

    // Three buyers, three pages -- the break is per note, not a one-off for the second.
    var three = Reports.BuildApprovalNotes(
        [(abcNote, abcLines), (xyzNote, xyzLines), (abcNote, abcLines)], head);
    var threePager = (DynamicDocumentPaginator)((IDocumentPaginatorSource)three).DocumentPaginator;
    threePager.PageSize = new Size(794, 1123);
    threePager.ComputePageCount();
    Check("MULTI - three buyers make three pages",
          threePager.PageCount == 3 && three.Blocks.OfType<Section>().Count(b => b.BreakPageBefore) == 2,
          $"{threePager.PageCount} page(s)");

    // ── ONE SHEET PER BUYER AND BROKER, whatever else differs ─────────────
    //
    // The grouping the PRINT uses, which is not the grouping the INVOICES use. DealKey carries
    // date, buyer, broker, broker %, terms and doc type, because two lines differing in any of
    // them cannot share an invoice -- and under it, one buyer bought through one broker on two
    // different terms came out as two pieces of paper for one delivery. A note asks a smaller
    // question: who is this parcel going to, through whom.
    var buyer = new PartyRef(1, "KIRAN EXPORTS", 30);
    var broker = new PartyRef(9, "JITESH SHAH", null, 1m);
    var grade = new Grade { GradeId = 1, Code = "COL", DisplayName = "Colour" };
    var size23 = new SizeBucket { SizeId = 1, Code = "+23" };
    var size18 = new SizeBucket { SizeId = 2, Code = "+18" };

    SaleLine Line(SizeBucket sz, int terms, string doc)
    {
        var l = new SaleLine { Grade = grade, Size = sz, GrossWeightCt = 1m, SelectionCt = 1m,
                               PricePerCt = 1000m };
        l.DealBuyer = buyer;
        l.DealBroker = broker;
        l.DealTermsDays = terms;
        l.DealDocType = doc;
        return l;
    }

    // Same buyer, same broker, DIFFERENT terms: two invoices, one sheet.
    var sameParty = new List<SaleLine> { Line(size23, 30, "BILL"), Line(size18, 60, "BILL") };

    Check("SHEET - two invoices to one buyer through one broker print as ONE note",
          InvoiceEntry.GroupForApproval(sameParty).Count == 1,
          $"{InvoiceEntry.GroupForApproval(sameParty).Count} note(s) from " +
          $"{InvoiceEntry.GroupDeals(sameParty).Count} deal(s)");

    Check("SHEET - and both parcels are on it",
          InvoiceEntry.GroupForApproval(sameParty)[0].Count == 2,
          $"{InvoiceEntry.GroupForApproval(sameParty)[0].Count} row(s)");

    // THE INVOICES ARE UNTOUCHED. Merging the sheet must not merge the documents: these two still
    // owe different money on different days, and one invoice cannot hold two sets of terms.
    Check("SHEET - but they are still two separate invoices",
          InvoiceEntry.GroupDeals(sameParty).Count == 2,
          "GroupDeals is not what changed");

    // A DIFFERENT BROKER is a different sheet, even for the same buyer -- the broker is on the
    // note's face ("Through ...") and two brokers cannot share one.
    var otherBroker = Line(size23, 30, "BILL");
    otherBroker.DealBroker = new PartyRef(7, "R K BROKER", null, 1m);
    Check("SHEET - a second broker for the same buyer is a second note",
          InvoiceEntry.GroupForApproval(new List<SaleLine> { sameParty[0], otherBroker }).Count == 2);

    // And no broker at all is its own group rather than joining whoever has one.
    var noBroker = Line(size18, 30, "BILL");
    noBroker.DealBroker = null;
    Check("SHEET - and a line with no broker does not join one that has",
          InvoiceEntry.GroupForApproval(new List<SaleLine> { sameParty[0], noBroker }).Count == 2);

    Console.WriteLine();
    Console.WriteLine(fail == 0 ? "All checks passed." : $"{fail} FAILED");
    Application.Current?.Shutdown();
});
t.SetApartmentState(ApartmentState.STA);
t.Start();
t.Join();
Environment.Exit(fail);
