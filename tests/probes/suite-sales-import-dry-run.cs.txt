// Dry run of the sales-import test sheet: parses it with the app's own SaleFileImport.Plan against
// Demo's LIVE catalogue and prints, per line, the bucket it resolves to and what it would deduct.
// READ ONLY -- Plan touches no database beyond reading grades and sizes, and nothing is imported.
//
// Run this BEFORE importing docs/templates/sales-import-stock-test.xlsx, and again after, to see
// where each spelling lands. A row resolving to "Dx1" rather than "NO DX" means the stray grade row
// is still there and supabase/merge_stray_dx1_into_no_dx.sql has not been run.

using System.IO;
using DiamondDesktop;
using DiamondDesktop.Data;
using InvProbe;

Db.Use(Db.Workspaces.First(w => w.Name == "Demo"));
Environment.SetEnvironmentVariable("SOLITAIRE_SESSION",
    Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                 "SolitaireDesk", $"session-{Db.Active.Ref}.dat"));
await ProbeEnv.SignInAsync();
await ProbeEnv.LoadProfileAsync();

var grades = await Repo.GradesAsync();
var sizes = await Repo.SizesAsync();
var gradeMap = SaleFileImport.AliasMap(grades.Select(g => (g.Code, g.Aliases)));
var sizeMap = SaleFileImport.SizeAliasMap(sizes.Select(s => s.Code));

string path = @"d:\diamond inventory\docs\templates\sales-import-stock-test.xlsx";
var plan = SaleFileImport.Plan(path, gradeMap, sizeMap);

Console.WriteLine($"file: {Path.GetFileName(path)}");
Console.WriteLine($"valid: {plan.IsValid}   invoices: {plan.Invoices.Count}   lines: {plan.LineCount}   "
                + $"skipped rows: {plan.SkippedRows}");

foreach (var p in plan.Problems)   Console.WriteLine($"   PROBLEM: {p.Message}");
foreach (var e in plan.Exceptions) Console.WriteLine($"   EXCEPTION: {e.Message}");
if (plan.UnknownGrades.Count > 0)
    Console.WriteLine($"   UNKNOWN GRADES (it would offer to create these): {string.Join(", ", plan.UnknownGrades)}");
if (plan.UnknownSizes.Count > 0)
    Console.WriteLine($"   UNKNOWN SIZES: {string.Join(", ", plan.UnknownSizes.Select(u => u.Label))}");

var pos = (await Db.Client.From<VStockPosition>().Get()).Models;

Console.WriteLine("\nplanned invoices, and what each line would take off the shelf:");
decimal totalOut = 0, totalShort = 0;
foreach (var inv in plan.Invoices)
{
    Console.WriteLine($"\n  {inv.InvoiceNo}  {inv.Date:dd-MM-yyyy}  {inv.Buyer}"
                    + $"  broker {(inv.Broker ?? "-")} {inv.BrokerPct}%  terms {inv.TermsDays}d  {inv.DocType}");
    foreach (var l in inv.Lines)
    {
        var b = pos.FirstOrDefault(p => p.GradeCode == l.GradeCode && p.SizeCode == l.SizeCode);
        decimal avail = b?.BalanceCt ?? 0;
        decimal takes = Math.Min(l.GrossCt, Math.Max(avail, 0));
        totalOut += takes; totalShort += l.GrossCt - takes;
        Console.WriteLine($"     row {l.ExcelRow}  ->  {l.GradeCode} x {l.SizeCode}"
                        + $"   gross {l.GrossCt,7:N4}   available {avail,9:N4}"
                        + $"   =>  deducts {takes,8:N4}, short {l.GrossCt - takes,7:N4}");
    }
    Console.WriteLine($"     invoice total {inv.Total:N2}");
}

Console.WriteLine($"\nexpected after import:  stock out {totalOut:N4} ct   short {totalShort:N4} ct");
Console.WriteLine("buyers created if missing: " + string.Join(", ", plan.Buyers));
Console.WriteLine("brokers: " + (plan.Brokers.Count == 0 ? "(none)" : string.Join(", ", plan.Brokers)));
