// NO DATABASE. Everything here is either a pure function or a control's state, so the whole of it
// can be checked without a connection -- which matters most for the memo rules, because the point
// of a memo is that it saves the state a stricter screen would have refused.
using System.Reflection;
using System.Threading;
using System.Windows;
using System.Windows.Controls;
using DiamondDesktop;
using DiamondDesktop.Data;

static IEnumerable<T> FindAll<T>(System.Windows.DependencyObject? d) where T : System.Windows.DependencyObject
{
    if (d is null) yield break;
    for (int i = 0; i < System.Windows.Media.VisualTreeHelper.GetChildrenCount(d); i++)
    {
        var c = System.Windows.Media.VisualTreeHelper.GetChild(d, i);
        if (c is T hit) yield return hit;
        foreach (var deep in FindAll<T>(c)) yield return deep;
    }
}

static T? FindDown<T>(System.Windows.DependencyObject? d) where T : System.Windows.DependencyObject
{
    if (d is null) return null;
    for (int i = 0; i < System.Windows.Media.VisualTreeHelper.GetChildrenCount(d); i++)
    {
        var c = System.Windows.Media.VisualTreeHelper.GetChild(d, i);
        if (c is T hit) return hit;
        if (FindDown<T>(c) is { } deep) return deep;
    }
    return null;
}

static string GetClickHandlerName(Button b)
{
    // The handler a XAML Click= wired up, read back off the button's own event table.
    var store = typeof(System.Windows.UIElement)
        .GetProperty("EventHandlersStore", BindingFlags.Instance | BindingFlags.NonPublic)!
        .GetValue(b);
    if (store is null) return "<none>";

    var got = store.GetType()
        .GetMethod("GetRoutedEventHandlers", BindingFlags.Instance | BindingFlags.Public)!
        .Invoke(store, [System.Windows.Controls.Primitives.ButtonBase.ClickEvent]);

    if (got is not System.Collections.IEnumerable seq) return "<none>";
    foreach (var item in seq)
    {
        var h = item?.GetType().GetProperty("Handler")?.GetValue(item) as Delegate;
        if (h is not null) return h.Method.Name;
    }
    return "<none>";
}

    if (!ok) fail++;
}

    // The Saved memos card is gone, and so is everything that fed it.
    Check("PRINT - the Saved memos card is gone",
          win.FindName("MemoCard") is null && win.FindName("MemoGrid") is null);
    Check("PRINT - and nothing is left calling for it",
          typeof(MainWindow).GetMethod("RefreshMemos", BindingFlags.Instance | BindingFlags.NonPublic) is null
          && typeof(MainWindow).GetMethod("FillMemosAsync", BindingFlags.Instance | BindingFlags.NonPublic) is null);

    // Confirm sale is untouched: it is still the one path that saves and posts, in that order.
    Check("PRINT - Confirm sale still writes, and is still the primary",
          ((Button)win.FindName("Post")!).Style == win.FindResource("DashGoldButton"));

    // ── Invoices: a memo is not a sale ─────────────────────────────────────
    var filter = (ComboBox)win.FindName("InvoiceStatusFilter")!;
    Check("LIST - the page opens on all sales",
          ((ComboBoxItem)filter.Items[0]).Content as string == "All sales" && filter.SelectedIndex == 0,
          $"index {filter.SelectedIndex}");
    // A filter that can only ever come back empty is worse than no filter.
    Check("LIST - and offers no Memo filter, because memos are never on this page",
          !filter.Items.OfType<ComboBoxItem>().Any(i => (i.Content as string) == "Memo"));

    typeof(MainWindow).GetField("_invoices", BindingFlags.Instance | BindingFlags.NonPublic)!
        .SetValue(win, new List<VInvoice>
        {
            new() { InvoiceId = 1, InvoiceNo = "INV-1", BuyerName = "ABC", Status = InvoiceStatus.POSTED },
            new() { InvoiceId = 2, InvoiceNo = null, BuyerName = "ABC", Status = InvoiceStatus.DRAFT },
            new() { InvoiceId = 3, InvoiceNo = "INV-3", BuyerName = "ABC", Status = InvoiceStatus.CANCELLED },
        });

    var apply = typeof(MainWindow).GetMethod("ApplyInvoiceFilter",
        BindingFlags.Instance | BindingFlags.NonPublic)!;
    var invoiceGrid = (DataGrid)win.FindName("InvoiceGrid")!;

    apply.Invoke(win, null);
    var shown = ((IEnumerable<VInvoice>)invoiceGrid.ItemsSource).ToList();
    Check("LIST - the memo is not in the list of sales",
          shown.Count == 2 && shown.All(i => i.Status != InvoiceStatus.DRAFT),
          string.Join(", ", shown.Select(i => i.Status)));
    Check("LIST - a cancelled sale still is: it happened",
          shown.Any(i => i.Status == InvoiceStatus.CANCELLED));

    // Excluded BEFORE the status filter, so no setting can bring one back.
    foreach (var item in filter.Items.OfType<ComboBoxItem>())
    {
        filter.SelectedItem = item;
        apply.Invoke(win, null);
        var rows = ((IEnumerable<VInvoice>)invoiceGrid.ItemsSource).ToList();
        if (rows.Any(i => i.Status == InvoiceStatus.DRAFT))
        {
            Check($"LIST - no setting brings a memo back ({item.Content})", false);
            break;
        }
    }
    Check("LIST - no setting of the filter brings a memo back", true);

    // ── a table that ends mid-row ends mid-BOX ────────────────────────────
    // The alias column is the one editable cell in the app, and EditableCell draws it as a real
    // input box. Where every other grid's clipped last row reads as "there is more below", a
    // half-drawn field reads as broken -- which is how it was reported.
    // The tab has to be SELECTED first: a TabControl does not realise an unselected tab's
    // content, so nothing is measured and SizeChanged never fires. The check would otherwise pass
    // an untouched MaxHeight of infinity and prove nothing.
    var tabs = (TabControl)win.FindName("Tabs")!;
    foreach (var t in tabs.Items.OfType<TabItem>())
        if (t.Header as string == "Master data") { t.IsSelected = true; break; }
    win.UpdateLayout();

    var gradeGrid = (DataGrid)win.FindName("GradeGrid")!;
    var gradeArea = (Panel)win.FindName("GradeArea")!;

    // The row has to be able to hold the box in the first place: 32 high inside a 4px margin,
    // over AppCell's 1px rule. If a restyle ever drops RowHeight below that, no amount of
    // snapping helps and this is the check that says so.
    Check("GRID - a grades row is tall enough for the field it draws",
          gradeGrid.RowHeight >= 32 + 4 + 4 + 1,
          $"row {gradeGrid.RowHeight}px for a 41px field");

    gradeArea.Height = 400;
    win.UpdateLayout();

    double chrome = gradeGrid.ColumnHeaderHeight + gradeGrid.BorderThickness.Top
                                                 + gradeGrid.BorderThickness.Bottom;
    double body = gradeGrid.MaxHeight - chrome;
    Check("GRID - and the table stops between rows, never through one",
          body > 0 && Math.Abs(body % gradeGrid.RowHeight) < 0.01,
          $"{body}px of body over {gradeGrid.RowHeight}px rows");

    // It must not overflow what it was given -- snapping DOWN to a whole row, not up past the card.
    Check("GRID - without spilling past the space the card gave it",
          gradeGrid.MaxHeight <= 400, $"{gradeGrid.MaxHeight}px inside 400px");

    // A card squeezed smaller than one row still shows a row and scrolls, rather than showing none.
    gradeArea.Height = 20;
    win.UpdateLayout();
    Check("GRID - a cramped card still shows one whole row",
          gradeGrid.MaxHeight >= chrome + gradeGrid.RowHeight,
          $"{gradeGrid.MaxHeight}px");
    gradeArea.Height = double.NaN;

    // ── a setting value is not always short ───────────────────────────────
    // The value column was pinned at 200px, which suits "15" and "60" and cuts stock_sheet_rates
    // off mid-list. It is a column, so it has to stay uniform down the card -- the fix is a share
    // of the width, not a per-row size.
    var settingRow = (DataTemplate)Application.Current.TryFindResource("SettingRow")!;
    var built = (FrameworkElement)settingRow.LoadContent();
    // The real SettingItem, not an anonymous type: Value binds TwoWay, and WPF throws outright on
    // a read-only property. Building the row against the type the screen actually uses is also the
    // only way this proves anything about the screen.
    built.DataContext = new SettingItem
    {
        Key = "stock_sheet_rates", Label = "Stock sheet rates", Description = "d",
        Category = "Other", OriginalValue = "",
        Value = "PREMIUM,+6.5,51000;PREMIUM,+11,48000;NO 1,+6.5,44000",
    };

    var cols = ((Grid)((Border)built).Child).ColumnDefinitions;
    Check("SETTING - the value column takes a share of the card, not a fixed 200px",
          cols[1].Width.IsStar, $"{cols[1].Width}");
    // Never narrower than it is today, and never the width of the page on a large monitor.
    Check("SETTING - with today's width as the floor and a ceiling above it",
          cols[1].MinWidth == 200 && cols[1].MaxWidth == 460,
          $"{cols[1].MinWidth}..{cols[1].MaxWidth}");

    // Wide card: the long value gets room the label was hogging.
    built.Width = 1100; built.Measure(new Size(1100, 200));
    built.Arrange(new Rect(0, 0, 1100, 200)); built.UpdateLayout();
    Check("SETTING - so a wide card gives a long value more than the old 200px",
          cols[1].ActualWidth > 200, $"{cols[1].ActualWidth:N0}px at 1100");

    // Narrow card: it holds the floor rather than collapsing.
    built.Width = 420; built.Measure(new Size(420, 200));
    built.Arrange(new Rect(0, 0, 420, 200)); built.UpdateLayout();
    Check("SETTING - and a narrow one does not squeeze it below the floor",
          cols[1].ActualWidth >= 200, $"{cols[1].ActualWidth:N0}px at 420");

    // A TextBox cannot ellipsise, so the tooltip is the whole overflow story for the values that
    // no sensible box will ever show in full.
    var valueBox = (TextBox)((Grid)((Border)built).Child).Children[1];
    Check("SETTING - a value too long for any box is still readable on hover",
          valueBox.ToolTip is not null);

    // ── a tooltip that fits on the screen ─────────────────────────────────
    // A tooltip is a Popup, so it was never clipped by a card. It laid a long value out as ONE
    // unwrapped line -- stock_sheet_rates is a whole sheet layout -- so it measured thousands of
    // pixels wide and ran off both edges with only its middle on screen. WPF already keeps a popup
    // inside the monitor's work area; it cannot fit something wider than the monitor.
    string longValue = string.Join(";", Enumerable.Range(0, 40).Select(i => $"PREMIUM,+6.5,{51000 + i}"));

    var tipStyle = (Style)Application.Current.FindResource(typeof(ToolTip));
    double maxW = (double)tipStyle.Setters.OfType<Setter>()
        .First(x => x.Property == FrameworkElement.MaxWidthProperty).Value;
    var tipTemplate = (DataTemplate)tipStyle.Setters.OfType<Setter>()
        .First(x => x.Property == ContentControl.ContentTemplateProperty).Value;

    Check("TIP - every tooltip in the app is bounded, not just the reported one",
          maxW > 0 && maxW <= 640, $"MaxWidth {maxW}");

    var tipText = (TextBlock)tipTemplate.LoadContent();
    Check("TIP - and its text wraps rather than running in one line",
          tipText.TextWrapping == TextWrapping.Wrap);

    // Measured through a real content host, not by poking the template's TextBlock directly: a
    // {Binding} on a detached element never resolves, and the check would have measured an empty
    // string and passed for the wrong reason.
    var tipHost = new ContentControl { Content = longValue, ContentTemplate = tipTemplate,
                                       MaxWidth = maxW };
    tipHost.Measure(new Size(double.PositiveInfinity, double.PositiveInfinity));

    // The whole bug, measured: a value 40 entries long has to come back inside the bound.
    Check("TIP - so a sheet-length value measures within the bound instead of off-screen",
          tipHost.DesiredSize.Width > 0 && tipHost.DesiredSize.Width <= maxW,
          $"{tipHost.DesiredSize.Width:N0}px of a possible {maxW:N0} for {longValue.Length} chars");

    // Bounded AND wrapped: width alone would clip the tail instead of showing it.
    var oneLine = new ContentControl { Content = "PREMIUM,+6.5,51000", ContentTemplate = tipTemplate };
    oneLine.Measure(new Size(double.PositiveInfinity, double.PositiveInfinity));
    Check("TIP - by flowing onto more lines, not by hiding the tail",
          tipHost.DesiredSize.Height > oneLine.DesiredSize.Height * 2,
          $"{tipHost.DesiredSize.Height:N0}px tall against {oneLine.DesiredSize.Height:N0} for one line");

    // The one tooltip whose content is a panel rather than a string has to opt out, or the
    // template above would render its type name.
    string tipMarkup = System.IO.File.ReadAllText(System.IO.Path.Combine(
        AppContext.BaseDirectory, "..", "..", "..", "..", "..", "DiamondDesktop", "MainWindow.xaml"));
    Check("TIP - the one composed tooltip opts out of the text template",
          tipMarkup.Contains("<ToolTip ContentTemplate=\"{x:Null}\">"),
          "dashboard chart row keeps its panel");

    // ── an invoice number is not a different invoice ──────────────────────
    // 118px cut "INV-2026-00004" to "INV-2026-0000", so two invoices on the card read the same.
    var drill = (DataGrid)win.FindName("DrillGrid")!;
    var invoiceNo = drill.Columns[0];
    Check("GRID - the invoice number column is sized from the numbers themselves",
          invoiceNo.Width.IsAuto, $"{invoiceNo.Width}");
    Check("GRID - with a floor that fits a full invoice number",
          invoiceNo.MinWidth >= 132, $"MinWidth {invoiceNo.MinWidth}");

    // ── a line goes under the last line, and stays there ──────────────────
    // Add line appends to Lines; a DataGrid sorts by default, so one click on a header reordered
    // the view and the appended row landed somewhere other than under the last line on screen.
    string entryMarkup = System.IO.File.ReadAllText(System.IO.Path.Combine(
        AppContext.BaseDirectory, "..", "..", "..", "..", "..", "DiamondDesktop", "MainWindow.xaml"));
    string modalMarkup = System.IO.File.ReadAllText(System.IO.Path.Combine(
        AppContext.BaseDirectory, "..", "..", "..", "..", "..", "DiamondDesktop", "UpdateInvoiceDialog.xaml"));

    Check("ORDER - the Sales entry grid cannot be re-sorted out of typed order",
          entryMarkup.Contains("CanUserSortColumns=\"False\""));
    Check("ORDER - and neither can the Update modal's",
          modalMarkup.Contains("CanUserSortColumns=\"False\""));

    var ordered = new InvoiceEntry();
    ordered.Lines.Clear();
    var first = new SaleLine { Remark = "first" };
    ordered.Lines.Add(first);
    var second = new SaleLine { Remark = "second" };
    ordered.Lines.Add(second);
    Check("ORDER - a new line goes after the last one, leaving the others alone",
          ReferenceEquals(ordered.Lines[0], first) && ReferenceEquals(ordered.Lines[^1], second),
          $"{ordered.Lines.Count} row(s), last is \"{ordered.Lines[^1].Remark}\"");

    win.Close();
    Console.WriteLine();
    Console.WriteLine(fail == 0 ? "All checks passed." : $"{fail} FAILED");
    Console.Out.Flush();
    Environment.Exit(fail);
});
t.SetApartmentState(ApartmentState.STA);
t.Start();
t.Join();
Environment.Exit(fail);
