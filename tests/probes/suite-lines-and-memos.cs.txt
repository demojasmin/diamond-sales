static IEnumerable<T> FindAll<T>(System.Windows.DependencyObject? d) where T : System.Windows.DependencyObject
{
    if (d is null) yield break;
    for (int i = 0; i < System.Windows.Media.VisualTreeHelper.GetChildrenCount(d); i++)
    {
        var c = System.Windows.Media.VisualTreeHelper.GetChild(d, i);
        if (c is T hit) yield return hit;
        foreach (var deep in FindAll<T>(c)) yield return deep;
    }
}

static T? FindDown<T>(System.Windows.DependencyObject? d) where T : System.Windows.DependencyObject
{
    if (d is null) return null;
    for (int i = 0; i < System.Windows.Media.VisualTreeHelper.GetChildrenCount(d); i++)
    {
        var c = System.Windows.Media.VisualTreeHelper.GetChild(d, i);
        if (c is T hit) return hit;
        if (FindDown<T>(c) is { } deep) return deep;
    }
    return null;
}

static string GetClickHandlerName(Button b)
{
    // The handler a XAML Click= wired up, read back off the button's own event table.
    var store = typeof(System.Windows.UIElement)
        .GetProperty("EventHandlersStore", BindingFlags.Instance | BindingFlags.NonPublic)!
        .GetValue(b);
    if (store is null) return "<none>";

    var got = store.GetType()
        .GetMethod("GetRoutedEventHandlers", BindingFlags.Instance | BindingFlags.Public)!
        .Invoke(store, [System.Windows.Controls.Primitives.ButtonBase.ClickEvent]);

    if (got is not System.Collections.IEnumerable seq) return "<none>";
    foreach (var item in seq)
    {
        var h = item?.GetType().GetProperty("Handler")?.GetValue(item) as Delegate;
        if (h is not null) return h.Method.Name;
    }
    return "<none>";
}

/// Puts a REAL buyer on an entry, the way the screen does. The deal is the line's now, so a
/// display string is not a buyer -- SelectedBuyer is what pushes a party onto every row.
static void WithBuyer(InvoiceEntry e, string name = "KIRAN EXPORTS", long id = 11)
{
    var party = e.Buyers.FirstOrDefault(b => b.Id == id);
    if (party is null) { party = new PartyRef(id, name); e.Buyers.Add(party); }
    e.SelectedBuyer = party;
}

void Check(string n, bool ok, string? d = null)
{
    Console.WriteLine($"{(ok ? "PASS" : "FAIL")}  {n}{(d is null ? "" : $"  . {d}")}");
    if (!ok) fail++;
}

    static bool HasWheelGuard(UIElement el)
    {
        var store = typeof(UIElement)
            .GetProperty("EventHandlersStore", BindingFlags.Instance | BindingFlags.NonPublic)!
            .GetValue(el);
        if (store is null) return false;
        var got = store.GetType()
            .GetMethod("GetRoutedEventHandlers", BindingFlags.Instance | BindingFlags.Public)!
            .Invoke(store, [UIElement.PreviewMouseWheelEvent]);
        if (got is not System.Collections.IEnumerable seq) return false;
        foreach (var item in seq)
            if ((item?.GetType().GetProperty("Handler")?.GetValue(item) as Delegate)?.Method.Name
                == "NoWheelChange") return true;
        return false;
    }

    static TextBox? EditPart(ComboBox cb) =>

    static string TypeInto(ComboBox cb, string word)
    {
        if (EditPart(cb) is not { } box) return "<no edit box>";

        box.Focus();
        System.Windows.Input.Keyboard.Focus(box);
        box.CaretIndex = box.Text.Length;

        // BOTH EVENTS, in InputManager's order: the TUNNELLING preview first, then the bubbling
        // one. RaiseEvent does only what it is asked, so a probe that sends the second alone never
        // runs a PreviewTextInput handler -- and two of the behaviours below live in one.

            var composition = new System.Windows.Input.TextComposition(
                System.Windows.Input.InputManager.Current, box, c.ToString());
            box.RaiseEvent(new System.Windows.Input.TextCompositionEventArgs(
                System.Windows.Input.InputManager.Current.PrimaryKeyboardDevice, composition)
            { RoutedEvent = UIElement.PreviewTextInputEvent });
            box.RaiseEvent(new System.Windows.Input.TextCompositionEventArgs(
                System.Windows.Input.InputManager.Current.PrimaryKeyboardDevice, composition)
            { RoutedEvent = UIElement.TextInputEvent });
        }

        return box.Text;
    }

          pe.VisibleLines.Count == 20 && ReferenceEquals(pe.VisibleLines[0], pe.Lines[20]));

    pe.Page = 2;
    Check("PAGE - and the last holds the remainder",
          pe.VisibleLines.Count == 5 && ReferenceEquals(pe.VisibleLines[0], pe.Lines[40]),
          $"{pe.VisibleLines.Count}");

    // Clamped, or a page number past the end shows an empty grid over a full invoice.
    pe.Page = 99;
    Check("PAGE - paging past the end lands on the last page", pe.Page == 2, $"{pe.Page}");
    pe.Page = -5;
    Check("PAGE - and before the start lands on the first", pe.Page == 0, $"{pe.Page}");

    // Nothing about paging reaches the invoice.
    pe.Buyer = "KIRAN EXPORTS";
    pe.Recalculate();
    Check("PAGE - the invoice keeps all 45 rows whatever page is shown",
          pe.RealLines.Count == 45 && pe.ToDraft(11, 1).Lines.Count == 45,
          $"{pe.ToDraft(11, 1).Lines.Count} line(s) in the payload");

    // Removing rows must not strand the page beyond the end.
    pe.Page = 2;
    for (int i = 0; i < 30; i++) pe.Lines.RemoveAt(pe.Lines.Count - 1);
    Check("PAGE - removing rows pulls the page back into range",
          pe.Page < pe.PageCount && pe.VisibleLines.Count > 0,
          $"page {pe.Page + 1} of {pe.PageCount}, {pe.VisibleLines.Count} row(s)");

    // Single page: the pager is furniture and stays hidden.
    var one = new InvoiceEntry();
    one.Repage();
    Check("PAGE - one page shows no pager", !one.HasPages && one.PageCount == 1);

    // ── the numbered pager ────────────────────────────────────────────────
    var np = new InvoiceEntry();
    np.Lines.Clear();
    for (int i = 0; i < 45; i++)
        np.Lines.Add(new SaleLine { Grade = noii, Size = s65, GrossWeightCt = 1m, SelectionCt = 1m, PricePerCt = 1000m });
    np.Repage();

    Check("PAGER - one numbered button per page",
          np.PageNumbers.Count == 3
          && np.PageNumbers.Select(c => c.Number).SequenceEqual(new[] { 1, 2, 3 }),
          string.Join(" ", np.PageNumbers.Select(c => c.Number)));
    Check("PAGER - and exactly one of them is the page you are on",
          np.PageNumbers.Count(c => c.Current) == 1 && np.PageNumbers[0].Current);
    Check("PAGER - the chevrons are spent at the first page, live at the other end",
          !np.CanPrevPage && np.CanNextPage);
    np.Page = 2;
    Check("PAGER - moving page moves the highlight with it",
          np.PageNumbers[2].Current && np.PageNumbers.Count(c => c.Current) == 1);
    Check("PAGER - and spends the other chevron", np.CanPrevPage && !np.CanNextPage);

    // A LONG ENTRY WINDOWS, or the numbers would push the chevrons off the card.
    for (int i = 0; i < 355; i++)
        np.Lines.Add(new SaleLine { Grade = noii, Size = s65, GrossWeightCt = 1m, SelectionCt = 1m, PricePerCt = 1000m });
    np.Page = 10;
    Check("PAGER - twenty pages show a window, not twenty buttons",
          np.PageCount == 20 && np.PageNumbers.Count == InvoiceEntry.PageWindow,
          $"{np.PageCount} pages, {np.PageNumbers.Count} buttons");
    Check("PAGER - centred on where you are",
          np.PageNumbers.Any(c => c.Current && c.Number == 11)
          && np.PageNumbers[0].Number == 8 && np.PageNumbers[^1].Number == 14,
          string.Join(" ", np.PageNumbers.Select(c => c.Number)));
    np.Page = 0;
    Check("PAGER - clamped at the start, so it never runs below page 1",
          np.PageNumbers[0].Number == 1 && np.PageNumbers.Count == InvoiceEntry.PageWindow,
          string.Join(" ", np.PageNumbers.Select(c => c.Number)));
    np.Page = 19;
    Check("PAGER - and at the end, so it never runs past the last",
          np.PageNumbers[^1].Number == 20 && np.PageNumbers.Count == InvoiceEntry.PageWindow,
          string.Join(" ", np.PageNumbers.Select(c => c.Number)));

    // ── rows per page ─────────────────────────────────────────────────────
    np.PageSize = 10;
    Check("PAGER - choosing 10 a page repages, and goes back to page 1",
          np.PageSize == 10 && np.VisibleLines.Count == 10 && np.Page == 0 && np.PageCount == 40,
          $"{np.VisibleLines.Count} rows, page {np.Page + 1} of {np.PageCount}");
    Check("PAGER - and 100 a page is four pages of the same rows",
          Resize(np, 100) == 4 && np.VisibleLines.Count == 100,
          $"{np.PageCount} page(s), {np.VisibleLines.Count} rows");
    Check("PAGER - the list offers the four the reference does",
          InvoiceEntry.PageSizes.SequenceEqual(new[] { 10, 20, 50, 100 }),
          string.Join(", ", InvoiceEntry.PageSizes));
    // NOTHING about paging reaches the invoice, at any page size.
    Check("PAGER - and all 400 rows are still on the invoice whatever page size is chosen",
          np.RealLines.Count == 400, $"{np.RealLines.Count}");
    np.PageSize = 20;

    static int Resize(InvoiceEntry e, int size) { e.PageSize = size; return e.PageCount; }

    // ── a tooltip that fits on the screen ─────────────────────────────────
    // A tooltip is a Popup, so it was never clipped by a card. It laid a long value out as ONE
    // unwrapped line -- stock_sheet_rates is a whole sheet layout -- so it measured thousands of
    // pixels wide and ran off both edges with only its middle on screen. WPF already keeps a popup
    // inside the monitor's work area; it cannot fit something wider than the monitor.
    string longValue = string.Join(";", Enumerable.Range(0, 40).Select(i => $"PREMIUM,+6.5,{51000 + i}"));

    var tipStyle = (Style)Application.Current.FindResource(typeof(ToolTip));
    double maxW = (double)tipStyle.Setters.OfType<Setter>()
        .First(x => x.Property == FrameworkElement.MaxWidthProperty).Value;
    var tipTemplate = (DataTemplate)tipStyle.Setters.OfType<Setter>()
        .First(x => x.Property == ContentControl.ContentTemplateProperty).Value;

    Check("TIP - every tooltip in the app is bounded, not just the reported one",
          maxW > 0 && maxW <= 640, $"MaxWidth {maxW}");

    var tipText = (TextBlock)tipTemplate.LoadContent();
    Check("TIP - and its text wraps rather than running in one line",
          tipText.TextWrapping == TextWrapping.Wrap);

    // Measured through a real content host, not by poking the template's TextBlock directly: a
    // {Binding} on a detached element never resolves, and the check would have measured an empty
    // string and passed for the wrong reason.
    var tipHost = new ContentControl { Content = longValue, ContentTemplate = tipTemplate,
                                       MaxWidth = maxW };
    tipHost.Measure(new Size(double.PositiveInfinity, double.PositiveInfinity));

    // The whole bug, measured: a value 40 entries long has to come back inside the bound.
    Check("TIP - so a sheet-length value measures within the bound instead of off-screen",
          tipHost.DesiredSize.Width > 0 && tipHost.DesiredSize.Width <= maxW,
          $"{tipHost.DesiredSize.Width:N0}px of a possible {maxW:N0} for {longValue.Length} chars");

    // Bounded AND wrapped: width alone would clip the tail instead of showing it.
    var oneLine = new ContentControl { Content = "PREMIUM,+6.5,51000", ContentTemplate = tipTemplate };
    oneLine.Measure(new Size(double.PositiveInfinity, double.PositiveInfinity));
    Check("TIP - by flowing onto more lines, not by hiding the tail",
          tipHost.DesiredSize.Height > oneLine.DesiredSize.Height * 2,
          $"{tipHost.DesiredSize.Height:N0}px tall against {oneLine.DesiredSize.Height:N0} for one line");

    // The one tooltip whose content is a panel rather than a string has to opt out, or the
    // template above would render its type name.
    string tipMarkup = System.IO.File.ReadAllText(System.IO.Path.Combine(
        AppContext.BaseDirectory, "..", "..", "..", "..", "..", "DiamondDesktop", "MainWindow.xaml"));
    Check("TIP - the one composed tooltip opts out of the text template",
          tipMarkup.Contains("<ToolTip ContentTemplate=\"{x:Null}\">"),
          "dashboard chart row keeps its panel");

    // ── an invoice number is not a different invoice ──────────────────────
    // 118px cut "INV-2026-00004" to "INV-2026-0000", so two invoices on the card read the same.
    var drill = (DataGrid)win.FindName("DrillGrid")!;
    var invoiceNo = drill.Columns[0];
    Check("GRID - the invoice number column is sized from the numbers themselves",
          invoiceNo.Width.IsAuto, $"{invoiceNo.Width}");
    Check("GRID - with a floor that fits a full invoice number",
          invoiceNo.MinWidth >= 132, $"MinWidth {invoiceNo.MinWidth}");

    // ── a line goes under the last line, and stays there ──────────────────
    // Add line appends to Lines; a DataGrid sorts by default, so one click on a header reordered
    // the view and the appended row landed somewhere other than under the last line on screen.
    string entryMarkup = System.IO.File.ReadAllText(System.IO.Path.Combine(
        AppContext.BaseDirectory, "..", "..", "..", "..", "..", "DiamondDesktop", "MainWindow.xaml"));
    string modalMarkup = System.IO.File.ReadAllText(System.IO.Path.Combine(
        AppContext.BaseDirectory, "..", "..", "..", "..", "..", "DiamondDesktop", "UpdateInvoiceDialog.xaml"));

    Check("ORDER - the Sales entry grid cannot be re-sorted out of typed order",
          entryMarkup.Contains("CanUserSortColumns=\"False\""));
    Check("ORDER - and neither can the Update modal's",
          modalMarkup.Contains("CanUserSortColumns=\"False\""));

    var ordered = new InvoiceEntry();
    ordered.Lines.Clear();
    var first = new SaleLine { Remark = "first" };
    ordered.Lines.Add(first);
    var second = new SaleLine { Remark = "second" };
    ordered.Lines.Add(second);
    Check("ORDER - a new line goes after the last one, leaving the others alone",
          ReferenceEquals(ordered.Lines[0], first) && ReferenceEquals(ordered.Lines[^1], second),
          $"{ordered.Lines.Count} row(s), last is \"{ordered.Lines[^1].Remark}\"");

    win.Close();
    Console.WriteLine();
    Console.WriteLine(fail == 0 ? "All checks passed." : $"{fail} FAILED");
    Console.Out.Flush();
    Environment.Exit(fail);
});
t.SetApartmentState(ApartmentState.STA);
t.Start();
t.Join();
Environment.Exit(fail);
