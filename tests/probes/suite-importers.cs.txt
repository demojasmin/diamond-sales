// Every importer, against real files, for the two cases that matter:
//
//   A · the file writes a sieve the catalogue already holds under ANOTHER notation.
//       Nothing may be offered, nothing may be created, and the numbers must not move.
//   B · the file writes a sieve the catalogue genuinely lacks.
//       It must be named, with the weight riding on it, and nothing may import silently short.
//
// No database and no hardcoded size list: the catalogue is built here, and which sizes each file
// uses is DISCOVERED by parsing it first. Rename a sieve on the paper and this still holds.
using System.IO;
using DiamondDesktop;
using DiamondDesktop.Data;

int fail = 0;
void Check(string n, bool ok, string? d = null)
{
    Console.WriteLine($"{(ok ? "PASS" : "FAIL")}  {n}{(d is null ? "" : $"  · {d}")}");
    if (!ok) fail++;
}

var grades = new (string Code, string? Aliases)[]
{
    ("NO 1","NO1;NO 1;1"), ("NO 1 BB","1BB;1 BB;NO1BB;NO 1BB"), ("NO 2","NO2;2;NO-2"),
    ("NO 2 BB","2BB;2 BB;NO2BB"), ("NO II","II;NOII;NO2SPOT"), ("NO DX","DX;NODX;DELUXE;NO-DX"),
    ("EX 1","EX1;EXTRA 1;E1;Ex1"), ("NO 3","NO3;3;NO-3"), ("NO 4","NO4;4;NO-4"),
    ("NO 5","NO5;5;NO-5"), ("NO 6","NO6;6;NO-6"), ("NO 7","NO7;7;NO-7"),
    ("TOP-COL","TOPCOL;TOP COL;TC;T COLOR"), ("COL","COLOUR;COLOR;C"), ("OW","OFFWHITE;OFF WHITE;O/W"),
    ("LC 1","LC1;L C 1;LC-1"), ("LC 2","LC2;L C 2;LC-2"), ("LC 3","LC3;L C 3;LC-3"),
    ("GH","GHOST;GH-VVS;GHVVS"), ("LB 1","LB1;L B 1;LB-1"), ("LB 2","LB2;L B 2;LB-2"),
    ("+14","14;PLUS14;P14"), ("EXTRA","EXTRA ASSORT;XTRA;ASSORT"), ("FL","F L;FL1"),
    ("1MB","1 MB;NO1MB;NO 1 MB"), ("-2 MB",",-2 MB;-2MB;,-2MB"), ("MIX", null),
}.Select((g, i) => new Grade { GradeId = i + 1, Code = g.Code, Aliases = g.Aliases, DisplayName = g.Code })
 .ToList();

// The catalogue as it stood when these files were WRITTEN. The mechanism tests below are about
// notation and detection, not about which sieves survive today, so they need a catalogue that
// actually covers the paper -- otherwise every assertion fails for the one uninteresting reason.
string[] full = ["-2", "-6.5", "+6.5", "+11", "14+", "+18", "+23", "0.2", "0.25", "1/6", "1/3"];

// The catalogue as it stands AFTER 0035: 14+ retired (so ImportableSizes drops it), +18 and +23
// deleted outright. Used at the end to show which of these files can still be imported today.
string[] today = ["-2", "-6.5", "+6.5", "+11", "0.2", "0.25", "1/6", "1/3"];
var gradeMap = SaleFileImport.AliasMap(grades.Select(g => (g.Code, g.Aliases)));

// The PDF sheets print the REPORT's grade labels -- "#", "DX1", "TOP co" -- not the catalogue
// codes, so the printed reader has its own map. Handing it the workbook's map loses nine lines
// and 808.96 ct of a sheet that is otherwise read perfectly.
var pdfGradeMap = MainWindow.PdfGradeLabelMap(grades);

string dir = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.UserProfile), "Downloads");
string At(string f) => Path.Combine(dir, f);
bool Have(string f) => File.Exists(At(f));

string Key(string c) => StockFileImport.SizeKey(c) ?? c;

// The same sieve, written the other way round. Derived, never tabulated: a signed code moves its
// sign to the far end, and a decimal that is the reciprocal of a whole number becomes that
// fraction. So "+6.5" gives "6.5+" and "0.2" gives "1/5" without either being named here.
static string Flip(string c)
{
    if (c.Length > 1 && (c[0] == '+' || c[0] == '-')) return c[1..] + c[0];
    if (decimal.TryParse(c, out decimal d) && d > 0m)
    {
        decimal inv = 1m / d;
        if (inv == decimal.Truncate(inv) && inv <= 64m) return "1/" + decimal.Truncate(inv);
    }
    return c;
}

Check("notation flip is derived, not tabulated",
      Flip("+6.5") == "6.5+" && Flip("-2") == "2-" && Flip("0.2") == "1/5" && Flip("0.25") == "1/4",
      $"{Flip("+6.5")} {Flip("-2")} {Flip("0.2")} {Flip("0.25")}");
Console.WriteLine();

// ── PDF stock ───────────────────────────────────────────────────────────────
foreach (string file in new[] { "JADU APR-AUG'26.pdf", "17-8-26.pdf" })
{
    if (!Have(file)) { Console.WriteLine($"SKIP  {file} not in Downloads\n"); continue; }
    Console.WriteLine($"── PDF · {file}");

    var baseline = PdfStockFile.Plan(At(file), pdfGradeMap, MainWindow.PdfSizeLabelMap(full));
    var used = baseline.SizeOrder.ToList();
    Check("PDF · reads against the full catalogue", baseline.IsValid,
          $"{baseline.TotalCarats:N2} ct, sizes {string.Join(",", used)}");
    Check("PDF · offers nothing to add", baseline.UnknownSizes.Count == 0);

    // A · every catalogue code written the other way round.
    var flipped = PdfStockFile.Plan(At(file), pdfGradeMap, MainWindow.PdfSizeLabelMap([.. full.Select(Flip)]));
    Check("PDF · A · still reads when the catalogue uses the other notation", flipped.IsValid);
    Check("PDF · A · to the same total, to the paisa",
          flipped.TotalCarats == baseline.TotalCarats, $"{flipped.TotalCarats:N2}");
    Check("PDF · A · and offers nothing to add",
          flipped.UnknownSizes.Count == 0, string.Join(",", flipped.UnknownSizes.Select(u => u.Label)));

    // B · drop the first sieve this sheet actually uses.
    if (used.Count > 0)
    {
        string drop = used[0];
        var missing = PdfStockFile.Plan(At(file), pdfGradeMap,
            MainWindow.PdfSizeLabelMap([.. full.Where(c => Key(c) != Key(drop))]));
        Check($"PDF · B · refuses when \"{drop}\" is not in the catalogue", !missing.IsValid);
        Check($"PDF · B · names \"{drop}\" as the sieve to add",
              missing.UnknownSizes.Any(u => Key(u.Label) == Key(drop)),
              string.Join(",", missing.UnknownSizes.Select(u => $"{u.Label}={u.Carats:N2}")));
        Check("PDF · B · and states the carats standing under it",
              missing.UnknownSizes.Any(u => Key(u.Label) == Key(drop) && u.Carats > 0m));
    }
    Console.WriteLine();
}

// ── Excel stock ─────────────────────────────────────────────────────────────
foreach (string file in new[] { "stk 13-08-26.xlsx", "stk BKC-JAN.xlsx" })
{
    if (!Have(file)) { Console.WriteLine($"SKIP  {file} not in Downloads\n"); continue; }
    Console.WriteLine($"── Excel stock · {file}");

    var baseline = StockFileImport.Plan(At(file), gradeMap, StockFileImport.SizeMap(full));
    var used = baseline.Rows.Select(r => r.SizeCode).Distinct().ToList();
    Check("XLS-STK · reads against the full catalogue", baseline.IsValid,
          $"{baseline.Rows.Count} rows, {baseline.TotalCarats:N2} ct, sizes {string.Join(",", used)}");
    Check("XLS-STK · offers nothing to add",
          baseline.UnknownSizes.Count == 0, string.Join(",", baseline.UnknownSizes.Select(u => u.Label)));

    var flipped = StockFileImport.Plan(At(file), gradeMap, StockFileImport.SizeMap([.. full.Select(Flip)]));
    Check("XLS-STK · A · still reads when the catalogue uses the other notation", flipped.IsValid);
    Check("XLS-STK · A · same rows, same carats",
          flipped.Rows.Count == baseline.Rows.Count && flipped.TotalCarats == baseline.TotalCarats,
          $"{flipped.Rows.Count} rows / {flipped.TotalCarats:N2} ct");
    Check("XLS-STK · A · and offers nothing to add",
          flipped.UnknownSizes.Count == 0, string.Join(",", flipped.UnknownSizes.Select(u => u.Label)));

    if (used.Count > 0)
    {
        string drop = used[0];
        var missing = StockFileImport.Plan(At(file), gradeMap,
            StockFileImport.SizeMap([.. full.Where(c => Key(c) != Key(drop))]));
        Check($"XLS-STK · B · names \"{drop}\" as the sieve to add",
              missing.UnknownSizes.Any(u => Key(u.Label) == Key(drop)),
              string.Join(",", missing.UnknownSizes.Select(u => $"{u.Label}={u.Carats:N2}")));
        Check("XLS-STK · B · with the weight that would otherwise vanish",
              missing.UnknownSizes.Sum(u => u.Carats) > 0m,
              $"{missing.UnknownSizes.Sum(u => u.Carats):N2} ct");
        Check("XLS-STK · B · and those carats are counted as unplaced, never dropped",
              missing.UnplacedCarats > 0m, $"{missing.UnplacedCarats:N2} ct unplaced");
        Check("XLS-STK · B · the sieve is offered ONCE however many rows use it",
              missing.UnknownSizes.Count(u => Key(u.Label) == Key(drop)) == 1);
        Check("XLS-STK · B · and the rows naming it are not imported",
              missing.Rows.Count < baseline.Rows.Count,
              $"{missing.Rows.Count} vs {baseline.Rows.Count}");
    }
    Console.WriteLine();
}

// ── the sale workbook offers to add a grade, like the two stock importers ──
// An unknown grade used to be reported as a skipped row and nothing more, leaving the user to
// open Master data, type the name themselves and run the import again. The two stock paths
// have offered to add it for a while; this is the sale path catching up.
foreach (string file in new[] { "Sale File Sample.xlsx", "Sale File - 5 months sample.xlsx" })
{
    if (!Have(file)) { Console.WriteLine($"SKIP  {file} - not in Downloads"); continue; }

    var baseline = SaleFileImport.Plan(At(file), gradeMap, SaleFileImport.SizeAliasMap(full));
    var used = baseline.Invoices.SelectMany(i => i.Lines).Select(l => l.GradeCode).Distinct().ToList();

    Check($"SALEGRADE · {file} offers nothing when every grade resolves",
          baseline.UnknownGrades.Count == 0, string.Join(",", baseline.UnknownGrades));

    if (used.Count == 0) continue;

    // Drop one grade the file actually uses. It must be NAMED, not merely skipped.
    string drop = used[0];
    var thin = SaleFileImport.AliasMap(
        grades.Where(g => g.Code != drop).Select(g => (g.Code, g.Aliases)));
    var missing = SaleFileImport.Plan(At(file), thin, SaleFileImport.SizeAliasMap(full));

    Check($"SALEGRADE · {file} names the grade it cannot place",
          missing.UnknownGrades.Count > 0, string.Join(",", missing.UnknownGrades));

    // Named once, however many rows use it -- the dialog lists names, not occurrences.
    Check($"SALEGRADE · {file} lists it once, not once per row",
          missing.UnknownGrades.Count == missing.UnknownGrades.Distinct(StringComparer.OrdinalIgnoreCase).Count(),
          $"{missing.UnknownGrades.Count} name(s), {missing.SkippedRows} row(s) skipped");

    // The name offered is the one the SHEET writes, so a grade created from it matches the next
    // workbook too. Never a tidied-up version, and never an existing display name.
    Check($"SALEGRADE · {file} offers the sheet's own spelling",
          missing.UnknownGrades.All(g => g.Trim().Length > 0 && g == g.Trim()),
          string.Join(" | ", missing.UnknownGrades.Take(3)));

    Check($"SALEGRADE · {file} still refuses to import those rows",
          missing.LineCount < baseline.LineCount,
          $"{missing.LineCount} vs {baseline.LineCount} lines");
}

// ── IMPORTING STOCK ASKS BEFORE IT DELETES ─────────────────────────────────
// Both stock importers now put the same question -- replace what is already imported, or add this
// file to it -- through the same AskReplaceOrAdd dialog. The workbook one used to answer it
// itself: it showed a confirmation that said the old parcels WOULD be deleted, and there was no
// other door. Asserted off the handler's source, because the path needs a file dialog and a
// database and a probe has neither.
string importer = File.ReadAllText(Path.Combine(
    AppContext.BaseDirectory, "..", "..", "..", "..", "..", "DiamondDesktop", "MainWindow.xaml.cs"));

int excelAt = importer.IndexOf("private async void ImportStock_Click", StringComparison.Ordinal);
int pdfAt = importer.IndexOf("private async void ImportStockPdf_Click", StringComparison.Ordinal);
string excel = importer[excelAt..importer.IndexOf("// ── Stock import · the printed sheet", excelAt, StringComparison.Ordinal)];
string pdf = importer[pdfAt..importer.IndexOf("private async Task RememberSheetSizesAsync", pdfAt, StringComparison.Ordinal)];

Check("STOCKMODE · the workbook importer asks whether to replace or add",
      excel.Contains("AskReplaceOrAdd("), "the same dialog the printed sheet uses");
Check("STOCKMODE · and the printed sheet still asks with it too",
      pdf.Contains("AskReplaceOrAdd("), "one dialog, one wording, two importers");

// BEFORE anything is written. An answer collected after the write would be a report, not a choice.
Check("STOCKMODE · the question comes before the write",
      excel.IndexOf("AskReplaceOrAdd(", StringComparison.Ordinal)
      < excel.IndexOf("Repo.ImportStockAsync(", StringComparison.Ordinal),
      "asked, then written");

// Cancel is a third answer and must write nothing at all.
Check("STOCKMODE · cancelling writes nothing",
      excel.Contains("if (choice == AppDialog.Answer.Cancel) { Say(\"Stock import cancelled\"); return; }"));

// The answer has to REACH the database. The workbook path used to call ImportStockAsync with no
// mode at all, which defaults to Replace -- so a choice that never got passed would look right on
// screen and delete the old parcels anyway.
Check("STOCKMODE · and the answer is what the write is given",
      excel.Contains("new Repo.StockImportMode(replace, Guid.NewGuid(), \"excel\")")
      && !System.Text.RegularExpressions.Regex.IsMatch(
             excel, @"ImportStockAsync\(plan\.Rows, asAt, gradeIds, sizeIds,\s*progressDialog\.Progress\)\)"),
      "no mode-less call left on this path");

// A database without 0027 has no batch list, so it can only replace -- and must say so honestly
// rather than offer a choice it cannot honour.
Check("STOCKMODE · a database that cannot add still gets the replace-only confirmation",
      excel.Contains("if (batches is null)") && excel.Contains("ConfirmStockImport(plan, picker.FileName, existing.Count, asAt)"),
      "pre-0027 falls back rather than lying");

// The dialog itself must offer both doors AND a way out, in the office's own words.
int askAt = importer.IndexOf("private AppDialog.Answer AskReplaceOrAdd", StringComparison.Ordinal);
string ask = importer[askAt..(askAt + 2400)];
Check("STOCKMODE · the dialog offers remove, keep, and cancel",
      ask.Contains("primaryText: \"Remove old & import latest\"")
      && ask.Contains("tertiaryText: \"Keep old & add latest\"")
      && ask.Contains("cancelText: \"Cancel\""),
      "three doors");
Check("STOCKMODE · and names what removing the old data actually costs",
      ask.Contains("Removing the old data deletes") && ask.Contains("Currently imported"),
      "the carats at stake, and which imports they are");

// ── OFFLINE · WHAT EACH IMPORTER DOES WITHOUT A CONNECTION ─────────────────
// Three buttons, three answers, and the differences are deliberate. Asserted off the source for the
// same reason as the block above: the path needs a file dialog, a network state and a database.
//
//   sales from Excel   refuses -- it needs the database to create buyers and brokers and hand back
//                      their ids, so the payload cannot even be built offline
//   stock from Excel   queues a replace
//   stock from PDF     queues a replace, since this change. It used to refuse outright.
string sales = importer[importer.IndexOf("private async void ImportExcel_Click", StringComparison.Ordinal)..excelAt];

Check("OFFLINE · the sales importer still refuses, and says why",
      sales.Contains("if (!Db.IsOnline)") && sales.Contains("Sales import needs a connection"),
      "buyer and broker ids cannot be invented client-side");

Check("OFFLINE · the printed sheet no longer refuses outright",
      !pdf.Contains("A stock sheet can only be imported while connected"),
      "it parks the count instead of losing it");

Check("OFFLINE · and parks it through the same queue the workbook uses",
      pdf.Contains("QueueStockImportAsync(plan, picker.FileName, asAt"),
      "one outbox, one guard, two importers");

// The WHOLE reason it used to refuse. Only a replace can be checked against a fingerprint before it
// replays; a queued APPEND would add a second copy of the same carats on reconnect. So the queue is
// reached BEFORE the replace-or-add dialog, never after it -- asking and then queueing a replace
// either way would make an ignored answer look honoured.
Check("OFFLINE · the replace-or-add question is skipped, not asked and ignored",
      pdf.IndexOf("QueueStockImportAsync(", StringComparison.Ordinal)
      < pdf.IndexOf("AskReplaceOrAdd(", StringComparison.Ordinal),
      "queued before the choice, so only a replace is ever parked");

// Reading the sheet offline needs the catalogue held in memory since sign-in. Without the fallback
// the read returns null and the button appears to do nothing at all.
Check("OFFLINE · it reads against the catalogue already in memory",
      pdf.Contains("grades = [.. Catalogue.Grades];")
      && pdf.Contains("The catalogue has not loaded yet"),
      "and says so plainly when there is none");

// Creating a grade or a sieve size is a WRITE. Offering to make one that cannot be made would end
// in a transport error over a dialog that had already said yes.
Check("OFFLINE · it does not offer to create grades or sizes it cannot write",
      pdf.Contains("if (Db.IsOnline\n            && plan.Problems.Count == 1")
      && pdf.Contains("if (Db.IsOnline && plan.UnknownSizes.Count > 0"),
      "the offer needs a connection, so it is only made on one");

Console.WriteLine(fail == 0 ? "All checks passed." : $"{fail} check(s) failed.");
Environment.Exit(fail == 0 ? 0 : 1);
